{{ h.filters.comment | escape }}
fragment half4 impressionistSubject(
    RasterizerData in [[stage_in]],
    texture2d<half> source [[texture(0)]],
    texture2d<half> subjectMask [[texture(1)]],
    constant float &brushSize [[buffer(0)]])
{
  constexpr sampler s(filter::linear, address::clamp_to_edge);

  half4 original = source.sample(s, in.uv);
  half  mask     = subjectMask.sample(s, in.uv).r;
  half4 painted  = impressionist(source, s, in.uv, brushSize);

  return mix(painted, original, mask);
}
